```prisma
model Coupon {
  id            String    @id @default(cuid())
  code          String    @unique
  name          String
  description   String?
  type          CouponType
  value         Float     // 折扣值或百分比
  minAmount     Float?    // 最低消费金额
  maxDiscount   Float?    // 最高折扣金额
  maxUses       Int?      // 最大使用次数
  usedCount     Int       @default(0)
  perUserLimit  Int       @default(1)
  startsAt      DateTime?
  expiresAt     DateTime?
  isActive      Boolean   @default(true)
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt

  usages        CouponUsage[]

  @@index([code])
  @@index([isActive])
  @@index([expiresAt])
}

model CouponUsage {
  id          String   @id @default(cuid())
  couponId    String
  userId      String
  orderId     String
  discount    Float    // 实际折扣金额
  usedAt      DateTime @default(now())

  coupon      Coupon   @relation(fields: [couponId], references: [id])
  order       Order    @relation(fields: [orderId], references: [id])

  @@index([couponId])
  @@index([userId])
}

enum CouponType {
  PERCENTAGE  // 百分比折扣 (如: 20% off)
  FIXED       // 固定金额折扣 (如: $10 off)
}
```

#### 任务 4: 会员积分体系 - 数据库

**工作量**: 2h
**负责人**: 后端
**依赖**: Prisma
**验收标准**: Points 模型创建

**具体步骤**:
1. 修改 `prisma/schema.prisma`
2. 添加 LoyaltyProgram 模型
3. 添加 CustomerPoints 模型
4. 添加 PointTransaction 模型
5. 执行 `prisma migrate dev`

**模型设计**:
```prisma
model LoyaltyProgram {
  id            String   @id @default(cuid())
  name          String   @default("默认积分计划")
  pointsRate    Float    @default(1.0)  // 每消费1元得积分
  pointsToYuan  Float    @default(0.01) // 100积分=1元
  minRedemption Int      @default(100) // 最低兑换积分
  isActive      Boolean  @default(true)
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt
}

model CustomerPoints {
  id            String   @id @default(cuid())
  userId        String   @unique
  balance       Int      @default(0)
  totalEarned   Int      @default(0)
  totalRedeemed Int      @default(0)
  tier          MemberTier @default(BRONZE)
  updatedAt     DateTime @updatedAt

  user          User     @relation(fields: [userId], references: [id])
  transactions  PointTransaction[]
}

model PointTransaction {
  id          String       @id @default(cuid())
  customerPointsId String
  userId      String
  amount      Int          // 正数=获得，负数=消耗
  type        PointType
  orderId     String?
  description String
  expiresAt   DateTime?    // 积分过期时间
  createdAt   DateTime     @default(now())

  customerPoints CustomerPoints @relation(fields: [customerPointsId], references: [id])

  @@index([userId])
  @@index([customerPointsId])
  @@index([type])
  @@index([expiresAt])
}

enum PointType {
  EARN       // 消费获得
  REDEEM     // 兑换抵扣
  BONUS      // 奖励
  EXPIRE     // 过期作废
  ADJUST     // 后台调整
  REFUND     // 退款扣回
}

enum MemberTier {
  BRONZE     // 青铜
  SILVER     // 白银
  GOLD       // 黄金
  PLATINUM   // 白金
}
```
