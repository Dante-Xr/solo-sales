```typescript
// 意图类型
type Intent =
  | "product_inquiry"      // 产品咨询
  | "order_status"         // 订单状态
  | "return_refund"        // 退换货
  | "shipping_info"         // 物流信息
  | "payment_issue"         // 支付问题
  | "complaint"             // 投诉
  | "general"              // 一般咨询

// 对话上下文
interface ConversationContext {
  sessionId: string
  userId?: string
  history: Message[]
  currentIntent?: Intent
  extractedEntities: {
    orderId?: string
    productId?: string
    email?: string
  }
  state: "initial" | "awaiting_confirmation" | "resolved"
}

// RAG 检索优化
interface RAGConfig {
  topK: 5                    // 检索 5 条相关文档
  similarityThreshold: 0.7   // 相似度阈值
  maxTokens: 500           // 回复最大 token
  temperature: 0.7          // 创造性
}
```

**模型设计**:
```prisma
model EmailSequence {
  id          String   @id @default(cuid())
  name        String
  trigger     TriggerType
  steps       EmailSequenceStep[]
  isActive    Boolean  @default(true)
  createdAt   DateTime @default(now())
}

model EmailSequenceStep {
  id            String   @id @default(cuid())
  sequenceId    String
  order         Int      // 执行顺序
  template      String
  delayHours    Int      // 延迟时间 (小时)
  condition     Json?    // 条件判断
  isActive      Boolean  @default(true)

  sequence      EmailSequence @relation(fields: [sequenceId], references: [id])
}

model EmailSequenceEnrollment {
  id          String   @id @default(cuid())
  sequenceId  String
  userId      String
  currentStep Int      @default(0)
  status      EnrollmentStatus
  enrolledAt  DateTime @default(now())
  completedAt DateTime?
}

enum TriggerType {
  ORDER_PLACED
  ORDER_SHIPPED
  ORDER_DELIVERED
  CART_ABANDONED
  PRODUCT_VIEWED
  CUSTOMER_INACTIVE
  BIRTHDAY
  FIRST_PURCHASE
}

enum EnrollmentStatus {
  ACTIVE
  COMPLETED
  UNENROLLED
  FAILED
}
```

**预置邮件序列**:
```
1. 订单确认序列:
   - T+0h: 订单确认邮件
   - T+0h: 发货通知
   - T+7d: 物流到达提醒
   - T+14d: 评价请求

2. 客户激活序列:
   - T+0h: 欢迎邮件
   - T+24h: 新手引导
   - T+7d: 首次购买激励
   - T+30d: 沉睡客户唤醒

3. 复购激励序列:
   - T+60d: 复购提醒
   - T+90d: 专属优惠
   - T+120d: 限时促销
```

## 五、风险与应对

### 5.1 主要风险

| 风险 | 可能性 | 影响 | 应对 |
|------|--------|------|------|
| RAG 客服响应质量不稳定 | 中 | 高 | 人工兜底，快速迭代 Prompt |

### 5.2 回滚方案

```bash
# 数据库回滚
prisma migrate rollback

# 代码回滚
git revert <commit>

# PWA 缓存清除
# 引导用户清除浏览器缓存
```
